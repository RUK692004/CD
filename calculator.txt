%}

%%

[0-9]+       { yylval = atoi(yytext); return NUMBER; }

[+\-*/()]    { return yytext[0]; }

[ \t]        ;

\n           { return '\n'; }

.            { return yytext[0]; }

%%

int yywrap()
{
    return 1;
}


//YACC



%{
#include <stdio.h>
#include <stdlib.h>

int yylex(void);
void yyerror(const char *s);
%}

%token NUMBER

%left '+' '-'
%left '*' '/'
%right UPLUS UMINUS

%%

input:
    expression '\n'
    {
        printf("Result = %d\n", $1);
        YYACCEPT;
    }
    ;

expression:
      NUMBER                 { $$ = $1; }
    | expression '+' expression { $$ = $1 + $3; }
    | expression '-' expression { $$ = $1 - $3; }
    | expression '*' expression { $$ = $1 * $3; }
    | expression '/' expression
      {
          if ($3 == 0)
          {
              printf("Division by zero!\n");
              YYABORT;
          }
          else
              $$ = $1 / $3;
      }
    | '-' expression %prec UMINUS { $$ = -$2; }
    | '+' expression %prec UPLUS  { $$ = $2; }
    | '(' expression ')'         { $$ = $2; }
    ;

%%

void yyerror(const char *s)
{
    printf("Invalid expression: %s\n", s);
}

int main()
{
    printf("Enter an arithmetic expression: ");
    yyparse();
    return 0;
}
