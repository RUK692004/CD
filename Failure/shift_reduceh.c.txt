#include <stdio.h>

char input[100], stack[100];
int top = -1, pos = 0;

int reduce() {
    if (top >= 0 && stack[top] == 'i') {
        stack[top] = 'E';
        return 1;
    }

    if (top >= 2 && stack[top] == ')' &&
        stack[top - 1] == 'E' && stack[top - 2] == '(') {
        top -= 2;
        stack[top] = 'E';
        return 1;
    }

    if (top >= 2 && stack[top] == 'E' &&
        stack[top - 1] == '*' && stack[top - 2] == 'E') {
        top -= 2;
        stack[top] = 'E';
        return 1;
    }

    if (top >= 2 && stack[top] == 'E' &&
        stack[top - 1] == '+' && stack[top - 2] == 'E' &&
        input[pos] != '*') {
        top -= 2;
        stack[top] = 'E';
        return 1;
    }

    return 0;
}

void show(const char *action) {
    stack[top + 1] = '\0';
    printf("%-12s %-12s %s\n", stack, input + pos, action);
}

int main() {
    printf("Enter expression using i: ");
    scanf("%99s", input);

    puts("STACK      INPUT      ACTION");

    while (input[pos]) {
        stack[++top] = input[pos++];
        show("SHIFT");

        while (reduce())
            show("REDUCE");
    }

    while (reduce())
        show("REDUCE");

    if (top == 0 && stack[0] == 'E')
        puts("Expression accepted");
    else
        puts("Expression rejected");

    return 0;
}
