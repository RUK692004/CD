//YACC


%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct node
{
    char data[20];
    struct node *left, *right;
};

struct node *createNode(char *value, struct node *l, struct node *r)
{
    struct node *n = malloc(sizeof(struct node));
    if (!n) exit(1);

    snprintf(n->data, sizeof(n->data), "%s", value);
    n->left = l;
    n->right = r;
    return n;
}

void printTree(struct node *root, int space)
{
    if (root == NULL) return;

    for (int i = 0; i < space; i++)
        printf("  ");

    printf("%s\n", root->data);

    printTree(root->left, space + 1);
    printTree(root->right, space + 1);
}

void freeTree(struct node *root)
{
    if (root == NULL) return;
    freeTree(root->left);
    freeTree(root->right);
    free(root);
}

int yylex(void);
void yyerror(const char *s);
%}

%union
{
    char *str;
    struct node *ptr;
}

%token <str> NUMBER ID
%type <ptr> expr term factor

%%

input:
    expr '\n'
    {
        printf("\nAbstract Syntax Tree:\n");
        printTree($1, 0);
        freeTree($1);
        YYACCEPT;
    }
    ;

expr:
      expr '+' term { $$ = createNode("+", $1, $3); }
    | expr '-' term { $$ = createNode("-", $1, $3); }
    | term          { $$ = $1; }
    ;

term:
      term '*' factor { $$ = createNode("*", $1, $3); }
    | term '/' factor { $$ = createNode("/", $1, $3); }
    | factor          { $$ = $1; }
    ;

factor:
      '(' expr ')' { $$ = $2; }
    | NUMBER {
          $$ = createNode($1, NULL, NULL);
          free($1);
      }
    | ID {
          $$ = createNode($1, NULL, NULL);
          free($1);
      }
    ;

%%

int main()
{
    printf("Enter an expression: ");
    yyparse();
    return 0;
}

void yyerror(const char *s)
{
    printf("Invalid expression\n");
}
